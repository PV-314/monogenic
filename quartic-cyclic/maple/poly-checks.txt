		r1LB:=rtsLB[i1]+rtsLB[i2]:
		r1UB:=rtsUB[i1]+rtsUB[i2]:
		
		d:=max(degree(r1LB,z),degree(r1UB,z)):
		if degree(r1LB,z)<>degree(r1UB,z) then
			print("BAD: degrees different, cannot bound r1+r2 when i1=",i1,", i2=",i2):
			print("degree(r1LB,z)=",degree(r1LB,z)," and degree(r1UB,z)=",degree(r1UB,z)):
			return():
		fi:
		r1_poly:=0:
		for i from 0 to d do
			if coeff(r1LB,z,i)<>coeff(r1UB,z,i) then
				print("BAD: coefficients different, cannot bound linear coeffient when i1=",i1,", i2=",i2):
				print("for i=",i,", coeff(r1LB,z,i)=",coeff(r1LB,z,i)," and coeff(r1UB,z,i)=",coeff(r1UB,z,i)):
				return():
			fi:
			r1_poly:=r1_poly+coeff(r1UB,z,i)*z^i:
		od:
		r1_poly:=sort(r1_poly,z):
		r1LB_rem:=r1LB-r1_poly:
		r1UB_rem:=r1UB-r1_poly:
		if degree(r1LB_rem)<>degree(r1UB_rem) then
			print("BAD: leading remainder term different when i1=",i1,", i2=",i2):
			print("r1LB_rem=",r1LB_rem,", r1UB_rem=",r1UB_rem):
			return():
		fi:
		if lcoeff(r1LB_rem)<>lcoeff(r1UB_rem) then

			print("r1LB_rem=",r1LB_rem,", r1UB_rem=",r1UB_rem):
			return():
		fi:

	local a0v,a1v,f,fa,fa1,g1,g2,g3,g4:

	fa:=sort(collect(subs(x=a1*x+a0,subs(z=zv,f)),x),x):
	for a1v from -301 to 301 do
	for a0v from -301 to 301 do
		fa1:=subs(a1=a1v,a0=a0v,fa):
		g1:=gcd(coeff(fa1,x,3),coeff(fa1,x,2)):
